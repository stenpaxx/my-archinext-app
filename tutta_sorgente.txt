import { useEffect, useState, type ComponentType } from 'react';
import { Menu, Save, LogIn, LogOut, Check } from 'lucide-react';

import { CockpitPanel } from '@/components/CockpitPanel';

import { getSupabase } from '@/lib/supabase';

  caricaDaSupabase,

} from '@/lib/supabaseSync';

type ModuleMap = Record<string, Record<string, ComponentType | undefined>>;

const allModuleFiles = import.meta.glob('./modules/*.tsx', { eager: true }) as ModuleMap;

function resolveModule(
  candidates: string[],
  exportNames: string[],
  label: string
): ComponentType {
  for (const path of candidates) {
    const mod = allModuleFiles[path];
    if (!mod) continue;
    for (const expName of exportNames) {
      if (typeof mod[expName] === 'function') {
        return mod[expName]!;
      }
    }
    if (typeof mod.default === 'function') {
      return mod.default;
    }
  }

  return () => (
    <div className="rounded-xl bg-white border border-zinc-300 p-5 text-xs text-zinc-700">
      Modulo <strong>{label}</strong> non ancora presente in <code>src/modules/</code>.
    </div>
  );
}

const AuditRAIModule = resolveModule(

export function App() {
  const { commessa } = useAttiva();
  const attivaId = useStore((s) => s.attivaId);
  const [activeSection, setActiveSection] = useState<SectionId>('due-diligence');
  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
  const [desktopCollapsed, setDesktopCollapsed] = useState(false);

  const [userEmail, setUserEmail] = useState<string | null>(null);
  const [authModalOpen, setAuthModalOpen] = useState(false);
  const [savedFeedback, setSavedFeedback] = useState(false);
  const [syncError, setSyncError] = useState<string | null>(null);

  const caricaDatiCloud = async () => {
    try {
      const cloudData = await caricaDaSupabase();
      if (cloudData && cloudData.commesse.length > 0) {
        useStore.setState((prev) => ({
          commesse: cloudData.commesse,
          attivaId: cloudData.commesse.some((c) => c.id === prev.attivaId)
            ? prev.attivaId
            : cloudData.commesse[0].id,
          vani: { ...prev.vani, ...cloudData.vani },
          voci: { ...prev.voci, ...cloudData.voci },
        }));
      }
    } catch {
      // mantiene i dati locali se offline
    }
  };

  useEffect(() => {
    const onResize = () => {
      if (window.innerWidth >= 1024) {
        setMobileMenuOpen(false);
      }
    };
    window.addEventListener('resize', onResize);
    return () => window.removeEventListener('resize', onResize);
  }, []);

  useEffect(() => {
    const sb = getSupabase();
    if (!sb) return;

    void sb.auth.getSession().then(({ data }) => {
      const mail = data.session?.user?.email ?? null;
      setUserEmail(mail);
      if (mail) {
        void caricaDatiCloud();
      }
    });

    const {
      data: { subscription },
    } = sb.auth.onAuthStateChange((_event, session) => {
      const mail = session?.user?.email ?? null;
      setUserEmail(mail);
      if (mail) {
        void caricaDatiCloud();
      }
    });

    return () => {
      subscription.unsubscribe();
    };
  }, []);

  const handleAuthButtonClick = async () => {
    if (userEmail) {
      const sb = getSupabase();
      if (sb) {
        try {
          await sb.auth.signOut();
        } catch {
          // ignora eventuali errori di rete
        }
      }
      setUserEmail(null);
      setSyncError(null);
    } else {
      setAuthModalOpen(true);
    }
  };

  const handleSave = async () => {
    setSyncError(null);
    const snapshot = useStore.getState();
    try {
      localStorage.setItem('my-archinext-manual-save', JSON.stringify(snapshot));
    } catch {
      // ignora errori storage
    }

    const commessaCorrente =
      snapshot.commesse.find((c) => c.id === snapshot.attivaId) ?? snapshot.commesse[0];

    if (userEmail && commessaCorrente) {
      try {
        await sincronizzaCommessaSuSupabase(
          commessaCorrente,
          snapshot.vani?.[commessaCorrente.id] ?? [],
          snapshot.voci?.[commessaCorrente.id] ?? []
        );
      } catch (err: unknown) {
        setSyncError(err instanceof Error ? err.message : 'Errore sync Cloud');
        return;
      }
    }

    setSavedFeedback(true);
    window.setTimeout(() => setSavedFeedback(false), 1800);
  };

  const renderSection = () => {
    const sectionKey = `${attivaId ?? 'default'}-${activeSection}`;
    switch (activeSection) {
      case 'due-diligence':
        return <DueDiligence key={sectionKey} />;
      case 'audit-rai':
        return <AuditRAIModule key={sectionKey} />;
      case 'abachi':
        return <AbachiModule key={sectionKey} />;
      case 'computo':
        return <ComputoModule key={sectionKey} />;
      case 'cronoprogramma':
        return <Cronoprogramma key={sectionKey} />;
      case 'bozze':
        return <Bozze key={sectionKey} />;
      case 'documenti':
        return <CassettoDocumentale key={sectionKey} />;
      case 'autolisp':
        return <GeneratoreLisp key={sectionKey} />;
      default:
        return <DueDiligence key={sectionKey} />;
    }
  };

  return (
    <div className="min-h-screen lg:h-screen lg:overflow-hidden bg-zinc-100 text-zinc-950 flex flex-col lg:flex-row">
      <Sidebar
        activeSection={activeSection}
        onSelectSection={setActiveSection}
        mobileOpen={mobileMenuOpen}
        onCloseMobile={() => setMobileMenuOpen(false)}
        desktopCollapsed={desktopCollapsed}
        onToggleDesktopCollapse={() => setDesktopCollapsed((v) => !v)}
      />

      <div className="flex-1 min-w-0 lg:h-screen lg:overflow-y-auto">
        <header className="sticky top-0 z-20 bg-zinc-950 lg:bg-zinc-100/95 text-white lg:text-zinc-950 backdrop-blur-xs border-b border-zinc-800 lg:border-zinc-200 px-3 sm:px-6 py-2.5 flex items-center justify-between gap-2">
          <div className="flex items-center gap-2 min-w-0 flex-1">
            {!mobileMenuOpen && (
              <button
                type="button"
                onClick={() => setMobileMenuOpen(true)}
                aria-label="Apri menu"
                className="lg:hidden inline-flex items-center gap-1.5 rounded-lg bg-white px-2.5 py-1.5 text-[11px] font-mono font-bold text-zinc-950 hover:bg-zinc-200 transition-colors shrink-0"
              >
                <Menu size={14} className="text-zinc-950" />
                <span>Menu</span>
              </button>
            )}

            {desktopCollapsed && (
              <button
                type="button"
                onClick={() => setDesktopCollapsed(false)}
                aria-label="Apri menu laterale"
                className="hidden lg:inline-flex items-center gap-1.5 rounded-lg bg-zinc-950 px-3 py-1.5 text-xs font-mono font-bold text-white hover:bg-zinc-800 transition-colors shrink-0"
              >
                <Menu size={14} />
                <span>Menu</span>
              </button>
            )}

            <div className="min-w-0 flex-1 truncate text-xs font-mono">
              <span className="font-bold">
                {commessa ? `${commessa.codice} — ${commessa.titolo}` : 'MY ARCHINEXT'}
              </span>
              {commessa?.indirizzo && (
                <span className="hidden md:inline text-zinc-500"> · {commessa.indirizzo}</span>
              )}
            </div>
          </div>

          <div className="flex items-center gap-1.5 sm:gap-2 shrink-0">
            {userEmail && (
              <span className="hidden sm:inline-block max-w-[180px] truncate text-[11px] font-mono text-zinc-300 lg:text-zinc-600">
                {userEmail}
              </span>
            )}

            <button
              type="button"
              onClick={() => void handleSave()}
              title={
                userEmail
                  ? 'Salva in locale e sincronizza su Supabase Cloud'
                  : 'Salva in locale (fai Login per sincronizzare su Cloud)'
              }
              className="inline-flex items-center gap-1.5 rounded-lg bg-emerald-600 px-2.5 py-1.5 text-[11px] font-mono font-bold text-white hover:bg-emerald-500 transition-colors shrink-0"
            >
              {savedFeedback ? <Check size={13} /> : <Save size={13} />}
              <span>{savedFeedback ? 'Saved' : 'Save'}</span>
            </button>

            <button
              type="button"
              onClick={() => void handleAuthButtonClick()}
              title={userEmail ? `Esci da ${userEmail}` : 'Accedi al Cloud'}
              className="inline-flex items-center gap-1.5 rounded-lg border border-zinc-600 lg:border-zinc-300 bg-zinc-900 lg:bg-white px-2.5 py-1.5 text-[11px] font-mono font-bold text-white lg:text-zinc-900 hover:bg-zinc-800 lg:hover:bg-zinc-200 transition-colors shrink-0"
            >
              {userEmail ? <LogOut size={13} /> : <LogIn size={13} />}
              <span>{userEmail ? 'Logout' : 'Login'}</span>
            </button>
          </div>
        </header>

        {syncError && (
          <div className="mx-auto max-w-[1600px] px-4 sm:px-6 pt-3">
            <div className="rounded-lg bg-red-50 border border-red-200 px-3 py-2 text-xs text-red-700 flex items-center justify-between">
              <span>Errore sincronizzazione Supabase: {syncError}</span>
              <button
                type="button"
                onClick={() => setSyncError(null)}
                className="font-mono underline ml-3"
              >
                Chiudi
              </button>
            </div>
          </div>
        )}

        <div className="mx-auto max-w-[1600px] p-4 sm:p-6 grid grid-cols-1 xl:grid-cols-12 gap-5 items-start">
          <main className="xl:col-span-8 min-w-0">{renderSection()}</main>
          <aside className="xl:col-span-4 min-w-0">
            <CockpitPanel
              activeSection={activeSection}
              onSelectSection={setActiveSection}
            />
          </aside>
        </div>
      </div>

      <AuthModal
        open={authModalOpen}
        onClose={() => setAuthModalOpen(false)}
        onSuccess={() => {
          const sb = getSupabase();
          if (sb) {
            void sb.auth.getSession().then(({ data }) => {
              const mail = data.session?.user?.email ?? null;
              setUserEmail(mail);
              if (mail) void caricaDatiCloud();
            });
          }
        }}
      />
    </div>
  );
}

export default App;

import { useState } from 'react';
import { Lock, Mail, Loader2, AlertCircle, CheckCircle2, X } from 'lucide-react';
import { getSupabase } from '@/lib/supabase';

interface Props {
  open: boolean;
  onClose: () => void;
  onSuccess?: () => void;
}

export function AuthModal({ open, onClose, onSuccess }: Props) {
  const [mode, setMode] = useState<'login' | 'signup'>('login');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);
  const [errorMsg, setErrorMsg] = useState('');
  const [infoMsg, setInfoMsg] = useState('');

  if (!open) return null;

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    setErrorMsg('');
    setInfoMsg('');

    try {
      const sb = getSupabase();
      if (!sb) throw new Error('Client Supabase non disponibile.');

      if (mode === 'login') {
        const { error } = await sb.auth.signInWithPassword({ email, password });
        if (error) throw error;
        onSuccess?.();
        onClose();
      } else {
        const { data, error } = await sb.auth.signUp({ email, password });
        if (error) throw error;
        if (data.session) {
          onSuccess?.();
          onClose();
        } else {
          setInfoMsg('Registrazione completata. Controlla la tua email per confermare l’account.');
        }
      }
    } catch (err: unknown) {
      setErrorMsg(err instanceof Error ? err.message : 'Errore di autenticazione');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="fixed inset-0 z-50 bg-slate-950/70 backdrop-blur-xs flex items-center justify-center p-4">
      <div className="bg-white text-slate-900 rounded-xl border border-slate-200 shadow-2xl w-full max-w-md overflow-hidden">
        <div className="px-5 py-4 bg-slate-900 text-white flex items-center justify-between">
          <div className="flex items-center gap-2">
            <Lock size={16} className="text-sky-400" />
            <h2 className="text-sm font-semibold">
              {mode === 'login' ? 'Accesso Professionista — My Archinext' : 'Crea Account Professionista'}
            </h2>
          </div>
          <button
            type="button"
            onClick={onClose}
            className="text-slate-400 hover:text-white p-1"
          >
            <X size={16} />
          </button>
        </div>

        <form onSubmit={handleSubmit} className="p-5 space-y-4">
          <div className="inline-flex w-full rounded-lg bg-slate-100 p-1">
            <button
              type="button"
              onClick={() => { setMode('login'); setErrorMsg(''); setInfoMsg(''); }}
              className={`flex-1 py-1.5 rounded-md text-xs font-medium transition-colors ${
                mode === 'login' ? 'bg-white text-slate-900 shadow-xs' : 'text-slate-600'
              }`}
            >
              Accedi
            </button>
            <button
              type="button"
              onClick={() => { setMode('signup'); setErrorMsg(''); setInfoMsg(''); }}
              className={`flex-1 py-1.5 rounded-md text-xs font-medium transition-colors ${
                mode === 'signup' ? 'bg-white text-slate-900 shadow-xs' : 'text-slate-600'
              }`}
            >
              Registrati
            </button>
          </div>

          <div className="space-y-1.5">
            <label className="block text-xs font-medium text-slate-700">Email professionale</label>
            <div className="relative">
              <Mail size={14} className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
              <input
                type="email"
                required
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                placeholder="nome@studioarchitettura.it"
                className="w-full pl-9 pr-3 py-2 text-xs border border-slate-300 rounded-lg focus:outline-none focus:border-sky-600"
              />
            </div>
          </div>

          <div className="space-y-1.5">
            <label className="block text-xs font-medium text-slate-700">Password</label>
            <div className="relative">
              <Lock size={14} className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
              <input
                type="password"
                required
                minLength={6}
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                placeholder="Minimo 6 caratteri"
                className="w-full pl-9 pr-3 py-2 text-xs border border-slate-300 rounded-lg focus:outline-none focus:border-sky-600"
              />
            </div>
          </div>

          {errorMsg && (
            <div className="flex items-center gap-2 text-xs px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700">
              <AlertCircle size={14} className="shrink-0" />
              <span>{errorMsg}</span>
            </div>
          )}

          {infoMsg && (
            <div className="flex items-center gap-2 text-xs px-3 py-2 rounded-lg bg-emerald-50 border border-emerald-200 text-emerald-800">
              <CheckCircle2 size={14} className="shrink-0" />
              <span>{infoMsg}</span>
            </div>
          )}

          <div className="flex items-center justify-end gap-2 pt-1">
            <button
              type="button"
              onClick={onClose}
              className="px-3.5 py-2 rounded-lg text-xs font-medium text-slate-600 hover:bg-slate-100"
            >
              Continua in locale
            </button>
            <button
              type="submit"
              disabled={loading}
              className="inline-flex items-center gap-2 px-4 py-2 rounded-lg text-xs font-semibold bg-sky-600 hover:bg-sky-700 disabled:opacity-50 text-white transition-colors"
            >
              {loading && <Loader2 size={14} className="animate-spin" />}
              <span>{mode === 'login' ? 'Accedi al Cloud' : 'Crea Account'}</span>
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

export function CatalogoAdminModal({ open, onClose, onUpdated }: Props) {
  const [tab, setTab] = useState<CatalogoTab>('regolamento');
  const [jsonText, setJsonText] = useState<string>(ESEMPIO_REC);
  const [saving, setSaving] = useState(false);
  const [status, setStatus] = useState<'idle' | 'success' | 'error'>('idle');
  const [message, setMessage] = useState<string>('');

  useEffect(() => {
    setStatus('idle');
    setMessage('');
    if (tab === 'regolamento') setJsonText(ESEMPIO_REC);
    if (tab === 'nta') setJsonText(ESEMPIO_NTA);
    if (tab === 'prezzari') setJsonText(ESEMPIO_PREZZARI);
  }, [tab]);

  if (!open) return null;

  const handleFileChange = async (file: File | undefined) => {
    if (!file) return;
    setStatus('idle');
    setMessage('');
    try {
      const content = await file.text();
      JSON.parse(content);
      setJsonText(content);
      setMessage(`File "${file.name}" caricato e validato.`);
      setStatus('success');
    } catch {
      setMessage('Il file selezionato non contiene JSON valido.');
      setStatus('error');
    }
  };

  const handleSave = async () => {
    setSaving(true);
    setStatus('idle');
    setMessage('');

    try {
      const supabase = getSupabase();
      if (!supabase) {
        throw new Error('Client Supabase non inizializzato in src/lib/supabase.ts.');
      }

      const parsed = JSON.parse(jsonText);
      const rows = Array.isArray(parsed) ? parsed : [parsed];
      if (rows.length === 0) {
        throw new Error('Il JSON non contiene elementi da importare.');
      }

      const table =
        tab === 'regolamento'
          ? 'catalogo_regolamento_edilizio'
          : tab === 'nta'
            ? 'catalogo_nta'
            : 'catalogo_prezzari';

      const onConflict =
        tab === 'regolamento'
          ? 'articolo'
          : tab === 'nta'
            ? 'sigla'
            : 'codice';

      const { error } = await supabase
        .from(table)
        .upsert(rows, { onConflict });

      if (error) {
        throw new Error(error.message);
      }

      setStatus('success');
      setMessage(`Aggiornate ${rows.length} voci su "${table}".`);
      onUpdated?.();
    } catch (err: unknown) {
      setStatus('error');
      setMessage(err instanceof Error ? err.message : 'Errore durante il salvataggio.');
    } finally {
      setSaving(false);
    }
  };

  return (
    <div className="fixed inset-0 z-50 bg-slate-950/70 backdrop-blur-xs flex items-center justify-center p-4">
      <div className="bg-white text-slate-900 rounded-xl border border-slate-200 shadow-2xl w-full max-w-3xl overflow-hidden">

              <h2 className="text-sm font-semibold">Cassetto Documentale Normativo</h2>
              <p className="text-[11px] text-slate-400">
                Aggiorna cataloghi REC Roma 2026, NTA PRG e Prezzari su Supabase senza toccare il codice
              </p>
            </div>

          <button
            type="button"
            onClick={onClose}
            className="text-slate-400 hover:text-white p-1 rounded transition-colors"
          >
            <X size={17} />
          </button>
        </div>

        <div className="p-5 space-y-4">
          <div className="flex flex-wrap items-center justify-between gap-3 border-b border-slate-200 pb-3">
            <div className="inline-flex rounded-lg bg-slate-100 p-1">
              <button
                type="button"
                onClick={() => setTab('regolamento')}
                className={`px-3 py-1.5 rounded-md text-xs font-medium transition-colors ${
                  tab === 'regolamento' ? 'bg-white text-slate-900 shadow-xs' : 'text-slate-600 hover:text-slate-900'
                }`}
              >

                className={`px-3 py-1.5 rounded-md text-xs font-medium transition-colors ${

                }`}
              >
                NTA PRG Roma

                className={`px-3 py-1.5 rounded-md text-xs font-medium transition-colors ${

import { useEffect, useState, useRef } from 'react';

export function CopyButton({ text, label = 'Copia' }: { text: string; label?: string }) {
  const [copied, setCopied] = useState(false);
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);

  useEffect(() => () => { if (timer.current) clearTimeout(timer.current); }, []);

  const copy = async () => {
    try {
      await navigator.clipboard.writeText(text);
    } catch {
      const ta = document.createElement('textarea');
      ta.value = text;
      document.body.appendChild(ta);
      ta.select();
      document.execCommand('copy');
      document.body.removeChild(ta);
    }
    setCopied(true);
    timer.current = setTimeout(() => setCopied(false), 2000);
  };

  return (
    <button
      onClick={copy}
      className={`inline-flex items-center gap-2 px-3 py-1.5 rounded text-xs font-medium transition-colors ${
        copied
          ? 'bg-emerald-700 text-white'
          : 'bg-slate-800 text-white hover:bg-slate-700'
      }`}
    >
      {copied ? 'Copiato!' : label}
    </button>
  );
}

import { useState } from 'react';
import { X, Lock, Mail, LogIn } from 'lucide-react';
import { getSupabase } from '@/lib/supabase';

interface LoginModalProps {
  open: boolean;
  onClose: () => void;
  onSuccessLocal: (label: string) => void;
}

export function LoginModal({ open, onClose, onSuccessLocal }: LoginModalProps) {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);
  const [errore, setErrore] = useState<string | null>(null);

  if (!open) return null;

  const sb = getSupabase();

  const handleSupabaseLogin = async (e: React.FormEvent) => {
    e.preventDefault();
    setErrore(null);

    if (!sb) {
      onSuccessLocal(email.trim() || 'Architetto Locale');
      onClose();
      return;
    }

    setLoading(true);
    try {
      const { error } = await sb.auth.signInWithPassword({
        email: email.trim(),
        password,
      });

      if (error) {
        setErrore(error.message);
        return;
      }

      onClose();
    } catch {
      setErrore('Errore durante il collegamento con Supabase Auth.');
    } finally {
      setLoading(false);
    }
  };

  const handleAccessoRapidoLocale = () => {
    onSuccessLocal(email.trim() || 'Direttore Tecnico (Locale)');
    onClose();
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 p-4">
      <div className="w-full max-w-sm rounded-2xl bg-white border border-zinc-300 p-5 shadow-xl space-y-4">
        <div className="flex items-center justify-between border-b border-zinc-200 pb-3">
          <div className="flex items-center gap-2">
            <Lock size={16} className="text-zinc-900" />

          <button
            type="button"

          >
            <X size={16} />
          </button>
        </div>

        <form onSubmit={(e) => void handleSupabaseLogin(e)} className="space-y-3">
          <label className="block space-y-1 text-xs">
            <span className="font-semibold text-zinc-700">Email</span>
            <div className="relative">

                size={14}
                className="absolute left-3 top-1/2 -translate-y-1/2 text-zinc-400"
              />

                type="email"

                value={email}

                placeholder="nome@studio.it"
                className="w-full rounded-lg border border-zinc-300 bg-white pl-9 pr-3 py-2 text-xs text-zinc-950 focus:outline-none focus:border-zinc-950"
              />
            </div>
          </label>

          <label className="block space-y-1 text-xs">
            <span className="font-semibold text-zinc-700">Password</span>
            <input
              type="password"
              required={Boolean(sb)}
              value={password}

              placeholder="••••••••"
              className="w-full rounded-lg border border-zinc-300 bg-white px-3 py-2 text-xs text-zinc-950 focus:outline-none focus:border-zinc-950"
            />
          </label>

          {errore && (

export async function fetchCataloghiNormativi() {
  const supabase = getSupabase();
  if (!supabase) {
    return {
      nta: [] as VoceNtaPrg[],
      regolamento: [] as VoceRegolamentoEdilizio[],
      prezzari: [] as VocePrezzario[],
      errors: {
        nta: 'Client Supabase non inizializzato',
        regolamento: 'Client Supabase non inizializzato',
        prezzari: 'Client Supabase non inizializzato',
      },
    };
  }

  const [ntaRes, recRes, prezzariRes] = await Promise.all([
    supabase.from('catalogo_nta').select('*').order('sigla'),
    supabase.from('catalogo_regolamento_edilizio').select('*').order('articolo'),
    supabase.from('catalogo_prezzari').select('*').eq('attivo', true).order('codice'),
  ]);

  return {
    nta: (ntaRes.data as VoceNtaPrg[]) || [],
    regolamento: (recRes.data as VoceRegolamentoEdilizio[]) || [],
    prezzari: (prezzariRes.data as VocePrezzario[]) || [],
    errors: {
      nta: ntaRes.error?.message || null,
      regolamento: recRes.error?.message || null,
      prezzari: prezzariRes.error?.message || null,
    },
  };
}

function parseNumero(valore: unknown, fallback = 0): number {
  if (typeof valore === 'number') {
    return Number.isFinite(valore) ? valore : fallback;
  }
  if (typeof valore === 'string') {
    const normalizzato = valore
      .replace(/\s+/g, ' ')
      .replace(/m²|mq|m2|mt|m/gi, '')
      .replace(',', '.')
      .trim();
    const match = normalizzato.match(/-?\d+(\.\d+)?/);
    if (match) {
      const n = parseFloat(match[0]);
      return Number.isFinite(n) ? n : fallback;
    }
  }
  return fallback;
}

    raw.h ??
    raw.hMedia ??
    raw.h_media ??
    raw.altezzaNetta ??
    raw.altezza_netta;

  const finCandidate =
    raw.superfFinestrata ??
    raw.superf_finestrata ??
    raw.superficieFinestrata ??
    raw.superficie_finestrata ??
    raw.finestrata ??
    raw.aeroilluminante ??
    raw.supFinestrata ??
    raw.sup_finestrata;

  const destCandidate =
    raw.destinazione ??
    raw.destinazioneUso ??
    raw.destinazione_uso ??
    raw.uso ??
    raw.tipo ??
    raw.category;

  const superficie = parseNumero(supCandidate, 0);
  const nome =
    typeof nomeCandidate === 'string' && nomeCandidate.trim()
      ? nomeCandidate.trim()
      : '';

  if (!nome && superficie <= 0) return null;

  const altezzaMedia = parseNumero(hCandidate, 2.7);
  const superfFinestrata = parseNumero(finCandidate, 0);
  const vmcRaw =
    raw.vmc ?? raw.aerazioneForzata ?? raw.aerazione_forzata ?? raw.ventilazione;
  const vmc =
    typeof vmcRaw === 'boolean'
      ? vmcRaw
      : typeof vmcRaw === 'string'
        ? /true|si|sì|vmc|forzata|meccanica|1/i.test(vmcRaw)
        : false;

  const destinazione =
    typeof destCandidate === 'string' && destCandidate.trim()
      ? destCandidate.trim()
      : /bagno|wc|servizio/i.test(nome)
        ? superfFinestrata > 0
          ? 'Bagno'
          : 'Bagno cieco'
        : /cucina|cottura/i.test(nome)
          ? 'Cucina abitabile'
          : /letto|camera|matrimoniale|singola/i.test(nome)
            ? 'Camera'
            : /soggiorno|pranzo|living|salone/i.test(nome)
              ? 'Soggiorno'
              : 'Residenziale';

  return {
    id:
      typeof raw.id === 'string' && raw.id.trim()
        ? raw.id.trim()
        : `synaps-${Date.now()}-${index}-${Math.random().toString(36).slice(2, 6)}`,
    nome: nome || `Vano Synaps ${index + 1}`,
    destinazione,
    superficie: Number(superficie.toFixed(2)),
    superfFinestrata: Number(superfFinestrata.toFixed(2)),
    altezzaMedia: Number((altezzaMedia > 0 ? altezzaMedia : 2.7).toFixed(2)),
    vmc,
    note:
      typeof raw.note === 'string' && raw.note.trim()
        ? raw.note.trim()
        : 'Importato da documento / output Synaps',
  };
}

function estraiVaniDaOggetto(data: unknown): Vano[] {
  const trovati: Vano[] = [];

  const visita = (nodo: unknown) => {
    if (!nodo) return;
    if (Array.isArray(nodo)) {
      for (const item of nodo) {
        if (item && typeof item === 'object' && !Array.isArray(item)) {
          const vano = normalizzaRecordInVano(
            item as Record<string, unknown>,
            trovati.length
          );
          if (vano) {
            trovati.push(vano);
            continue;
          }
        }
        visita(item);
      }
      return;
    }

    if (typeof nodo === 'object') {
      const obj = nodo as Record<string, unknown>;
      const chiaviVani = [
        'vani',
        'rooms',
        'ambienti',
        'locali',
        'spazi',
        'planimetria',
        'risultati',
        'data',
      ];
      for (const chiave of chiaviVani) {
        if (obj[chiave] !== undefined) {
          visita(obj[chiave]);
        }
      }
    }
  };

  visita(data);
  return trovati;
}

function estraiVaniDaTesto(testo: string): Vano[] {
  const pulito = testo.trim();
  if (!pulito) return [];

  if (pulito.startsWith('{') || pulito.startsWith('[')) {
    try {
      const parsed = JSON.parse(pulito);
      const daJson = estraiVaniDaOggetto(parsed);
      if (daJson.length > 0) return daJson;
    } catch {
      // Prosegue con parse testuale/CSV
    }
  }

  const righe = pulito
    .split(/\r?\n/)
    .map((r) => r.trim())
    .filter(Boolean);

  if (righe.length === 0) return [];

  const separatore = righe[0].includes(';')
    ? ';'
    : righe[0].includes('\t')
      ? '\t'
      : righe[0].includes('|')
        ? '|'
        : ',';

  const colonne = righe[0].split(separatore).map((c) => c.trim().toLowerCase());
  const haIntestazione = colonne.some((c) =>
    /nome|vano|room|sup|mq|area|consistenza|altezza|h\b|finestr|destinazione/.test(c)
  );

  const vani: Vano[] = [];
  const startIdx = haIntestazione ? 1 : 0;

  for (let i = startIdx; i < righe.length; i++) {
    const parti = righe[i].split(separatore).map((p) => p.trim());
    if (parti.length < 2) continue;

    if (haIntestazione) {
      const record: Record<string, string> = {};
      colonne.forEach((col, cIdx) => {
        record[col] = parti[cIdx] ?? '';
      });
      const vano = normalizzaRecordInVano(record, vani.length);
      if (vano) vani.push(vano);
    } else {
      const nome = parti[0];
      const supMatch = righe[i].match(/(\d+(?:[.,]\d+)?)\s*(?:m²|mq|m2)\b/i);
      const hMatch = righe[i].match(
        /\b(?:h|alt(?:ezza)?)\s*[:=]?\s*(\d+(?:[.,]\d+)?)\s*m?\b/i
      );
      const finMatch = righe[i].match(
        /\b(?:fin(?:estrata)?|rai|apribile)\s*[:=]?\s*(\d+(?:[.,]\d+)?)\s*(?:m²|mq|m2)?\b/i
      );

      const superficie = supMatch
        ? parseNumero(supMatch[1], 0)
        : parseNumero(parti[1], 0);
      const altezzaMedia = hMatch
        ? parseNumero(hMatch[1], 2.7)
        : parseNumero(parti[2], 2.7);
      const superfFinestrata = finMatch
        ? parseNumero(finMatch[1], 0)
        : parseNumero(parti[3], 0);

      const vano = normalizzaRecordInVano(
        {
          nome,
          destinazione: parti.find((p) =>
            /soggiorno|cucina|camera|letto|bagno|wc|studio|ingresso|corridoio|disimpegno/i.test(
              p
            )
          ),
          superficie,
          altezzaMedia: altezzaMedia > 0 ? altezzaMedia : 2.7,
          superfFinestrata,
          vmc: /vmc|aerazione forzata|meccanica/i.test(righe[i]),
        },
        vani.length
      );
      if (vano) vani.push(vano);
    }
  }

  return vani;
}

function generaProssimoCodice(commesse: Commessa[]): string {
  const anno = new Date().getFullYear();
  const progressivi = commesse
    .map((c) => {
      const match = c.codice.match(/(\d+)$/);
      return match ? parseInt(match[1], 10) : 0;
    })
    .filter((n) => !Number.isNaN(n));

  const prossimo = (progressivi.length ? Math.max(...progressivi) : 14) + 1;
  return `${anno}-${String(prossimo).padStart(3, '0')}`;
}

export const useStore = create<ArchinextState>()(
  persist(
    (set, get) => {
      const applicaVaniImportati = (
        vaniEstratti: Vano[],
        options?: { commessaId?: string; modalita?: 'sostituisci' | 'unisci' }
      ): RisultatoImportSynaps => {
        const targetId = options?.commessaId || get().attivaId;
        const modalita = options?.modalita ?? 'sostituisci';

        if (!targetId) {
          return {
            ok: false,
            vaniImportati: 0,
            superficieTotale: 0,
            messaggio: 'Nessuna commessa attiva selezionata.',
            vani: [],
          };
        }

        if (vaniEstratti.length === 0) {
          return {
            ok: false,
            vaniImportati: 0,
            superficieTotale: 0,
            messaggio:
              'Nessun vano riconosciuto nel file. Verifica che siano presenti nome/ambiente, superficie (mq) e altezza.',
            vani: [],
          };
        }

        let listaFinale = vaniEstratti;
        set((state) => {
          const attuali = state.vani[targetId] ?? [];
          listaFinale =
            modalita === 'unisci' ? [...attuali, ...vaniEstratti] : vaniEstratti;
          return {
            vani: {
              ...state.vani,
              [targetId]: listaFinale,
            },
          };
        });

        const superficieTotale = Number(
          listaFinale
            .reduce((acc, v) => acc + (Number(v.superficie) || 0), 0)
            .toFixed(2)
        );

        return {
          ok: true,
          vaniImportati: vaniEstratti.length,
          superficieTotale,
          messaggio: `Importati ${vaniEstratti.length} vani (${superficieTotale.toFixed(2)} mq totali).`,
          vani: listaFinale,
        };
      };

      const handleImportaDaSynaps = async (
        fileOrPayload:
          | File
          | string
          | Record<string, unknown>
          | Array<Record<string, unknown>>,
        options?: { commessaId?: string; modalita?: 'sostituisci' | 'unisci' }
      ): Promise<RisultatoImportSynaps> => {
        if (typeof File !== 'undefined' && fileOrPayload instanceof File) {
          const contenuto = await fileOrPayload.text();
          const estratti = estraiVaniDaTesto(contenuto);
          return applicaVaniImportati(estratti, options);
        }

        if (typeof fileOrPayload === 'string') {
          const estratti = estraiVaniDaTesto(fileOrPayload);
          return applicaVaniImportati(estratti, options);
        }

        const estratti = estraiVaniDaOggetto(fileOrPayload);
        return applicaVaniImportati(estratti, options);
      };

      const handleAggiungiVano = (
        commessaIdOrVano?: string | Partial<Vano>,
        maybeVano?: Partial<Vano>
      ) => {
        const targetId =
          typeof commessaIdOrVano === 'string' ? commessaIdOrVano : get().attivaId;
        const dati =
          typeof commessaIdOrVano === 'object' && commessaIdOrVano !== null
            ? commessaIdOrVano
            : maybeVano;
        if (!targetId) return;

        const esistenti = get().vani[targetId] ?? [];
        const nuovoVano: Vano = {
          id: dati?.id || `vano-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
          nome: dati?.nome?.trim() || `Vano ${esistenti.length + 1}`,
          destinazione: dati?.destinazione?.trim() || 'Soggiorno',
          superficie: dati?.superficie !== undefined ? Number(dati.superficie) : 14,
          superfFinestrata:
            dati?.superfFinestrata !== undefined ? Number(dati.superfFinestrata) : 2.0,
          altezzaMedia:
            dati?.altezzaMedia !== undefined ? Number(dati.altezzaMedia) : 3.0,
          vmc: Boolean(dati?.vmc),
          note: dati?.note,
        };

        set((state) => ({
          vani: {
            ...state.vani,
            [targetId]: [...(state.vani[targetId] ?? []), nuovoVano],
          },
        }));
      };

      const handleAggiornaVano = (
        commessaIdOrVanoId: string,
        vanoIdOrPatch: string | Partial<Vano>,
        maybePatch?: Partial<Vano>
      ) => {
        const hasExplicitCommessa = typeof vanoIdOrPatch === 'string';
        const targetId = hasExplicitCommessa ? commessaIdOrVanoId : get().attivaId;
        const vanoId = hasExplicitCommessa ? vanoIdOrPatch : commessaIdOrVanoId;
        const patch = (hasExplicitCommessa ? maybePatch : vanoIdOrPatch) ?? {};
        if (!targetId || !vanoId) return;

        set((state) => ({
          vani: {
            ...state.vani,
            [targetId]: (state.vani[targetId] ?? []).map((v) =>
              v.id === vanoId ? { ...v, ...patch } : v
            ),
          },
        }));
      };

      const handleRimuoviVano = (commessaIdOrVanoId: string, maybeVanoId?: string) => {
        const hasExplicitCommessa = typeof maybeVanoId === 'string';
        const targetId = hasExplicitCommessa ? commessaIdOrVanoId : get().attivaId;
        const vanoId = hasExplicitCommessa ? maybeVanoId! : commessaIdOrVanoId;
        if (!targetId || !vanoId) return;

        set((state) => ({
          vani: {
            ...state.vani,
            [targetId]: (state.vani[targetId] ?? []).filter((v) => v.id !== vanoId),
          },
        }));
      };

      const normalizzaVoce = (
        dati: Partial<VoceComputo> | undefined,
        fallbackId: string
      ): VoceComputo => {
        const um = dati?.unitaMisura?.trim() || dati?.um?.trim() || 'mq';
        const qta = dati?.quantita !== undefined ? Number(dati.quantita) : 1;
        const prezzo =
          dati?.prezzoUnitario !== undefined
            ? Number(dati.prezzoUnitario)
            : dati?.prezzo !== undefined
              ? Number(dati.prezzo)
              : 35;
        const sicurezza = Boolean(dati?.oneriSicurezza ?? dati?.sicurezza);

        return buildVoce(
          dati?.id || fallbackId,
          dati?.codice?.trim() || 'A 01.01.001.a',
          dati?.descrizione?.trim() || 'Nuova lavorazione Prezzario Lazio',
          um,
          qta,
          prezzo,
          sicurezza,
          dati?.categoria?.trim() || 'Opere edili'
        );
      };

      const handleSetVoci = (
        commessaId: string,
        nuoveVoci: Array<Partial<VoceComputo>>
      ) => {
        const targetId = commessaId || get().attivaId;
        if (!targetId) return;

        const normalizzate = (nuoveVoci ?? []).map((item, idx) =>
          normalizzaVoce(
            item,
            item.id || `voce-${Date.now()}-${idx}-${Math.random().toString(36).slice(2, 5)}`
          )
        );

        set((state) => ({
          voci: {
            ...state.voci,
            [targetId]: normalizzate,
          },
        }));
      };

      const handleAggiungiVoce = (
        commessaIdOrVoce?: string | Partial<VoceComputo>,
        maybeVoce?: Partial<VoceComputo>
      ) => {
        const targetId =
          typeof commessaIdOrVoce === 'string' ? commessaIdOrVoce : get().attivaId;
        const dati =
          typeof commessaIdOrVoce === 'object' && commessaIdOrVoce !== null
            ? commessaIdOrVoce
            : maybeVoce;
        if (!targetId) return;

        const nuovaVoce = normalizzaVoce(
          dati,
          `voce-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`
        );

        set((state) => ({
          voci: {
            ...state.voci,
            [targetId]: [...(state.voci[targetId] ?? []), nuovaVoce],
          },
        }));
      };

      const handleAggiornaVoce = (
        commessaIdOrVoceId: string,
        voceIdOrPatch: string | Partial<VoceComputo>,
        maybePatch?: Partial<VoceComputo>
      ) => {
        const hasExplicitCommessa = typeof voceIdOrPatch === 'string';
        const targetId = hasExplicitCommessa ? commessaIdOrVoceId : get().attivaId;
        const voceId = hasExplicitCommessa ? voceIdOrPatch : commessaIdOrVoceId;
        const patch = (hasExplicitCommessa ? maybePatch : voceIdOrPatch) ?? {};
        if (!targetId || !voceId) return;

        set((state) => ({
          voci: {
            ...state.voci,
            [targetId]: (state.voci[targetId] ?? []).map((item) => {
              if (item.id !== voceId) return item;
              const nextUm = patch.unitaMisura ?? patch.um ?? item.unitaMisura;
              const nextQta =
                patch.quantita !== undefined ? Number(patch.quantita) : item.quantita;
              const nextPrezzo =
                patch.prezzoUnitario !== undefined
                  ? Number(patch.prezzoUnitario)
                  : patch.prezzo !== undefined
                    ? Number(patch.prezzo)
                    : item.prezzoUnitario;
              const nextSic =
                patch.oneriSicurezza !== undefined
                  ? Boolean(patch.oneriSicurezza)
                  : patch.sicurezza !== undefined
                    ? Boolean(patch.sicurezza)
                    : item.oneriSicurezza;

              return {
                ...item,
                ...patch,
                categoria: patch.categoria ?? item.categoria ?? 'Opere edili',
                unitaMisura: nextUm,
                um: nextUm,
                quantita: nextQta,
                prezzoUnitario: nextPrezzo,
                prezzo: nextPrezzo,
                importo: Number((nextQta * nextPrezzo).toFixed(2)),
                oneriSicurezza: nextSic,
                sicurezza: nextSic,
              };
            }),
          },
        }));
      };

      const handleRimuoviVoce = (commessaIdOrVoceId: string, maybeVoceId?: string) => {
        const hasExplicitCommessa = typeof maybeVoceId === 'string';
        const targetId = hasExplicitCommessa ? commessaIdOrVoceId : get().attivaId;
        const voceId = hasExplicitCommessa ? maybeVoceId! : commessaIdOrVoceId;
        if (!targetId || !voceId) return;

        set((state) => ({
          voci: {
            ...state.voci,
            [targetId]: (state.voci[targetId] ?? []).filter((item) => item.id !== voceId),
          },
        }));
      };

      const handlePopolaLazio = (commessaId?: string) => {
        const targetId = typeof commessaId === 'string' ? commessaId : get().attivaId;
        if (!targetId) return;

        const generate: VoceComputo[] = VOCI_TIPO_LAZIO.map((v, i) =>
          buildVoce(
            `laz-${Date.now()}-${i}-${Math.random().toString(36).slice(2, 5)}`,
            v.codice,
            v.descrizione,
            v.unitaMisura,
            v.quantita,
            v.prezzoUnitario,
            Boolean(v.oneriSicurezza),
            v.categoria
          )
        );

        set((state) => ({
          voci: {
            ...state.voci,
            [targetId]: generate,
          },
        }));
      };

      const handleImportaDaDocumento = async (
        file: File,
        options?: {
          commessaId?: string;
          categoria?: string;
          modalita?: 'sostituisci' | 'unisci';
        }
      ): Promise<RisultatoImportDocumento> => {
        const targetId = options?.commessaId || get().attivaId;
        if (!targetId) {
          return {
            vaniAggiunti: 0,
            vociAggiornate: 0,
            riepilogo: 'Nessuna commessa attiva.',
          };
        }

        let testo = '';
        try {
          testo = await file.text();
        } catch {
          testo = '';
        }

        const estratti = testo ? estraiVaniDaTesto(testo) : [];
        let vaniAggiunti = 0;

        if (estratti.length > 0) {
          const res = applicaVaniImportati(estratti, {
            commessaId: targetId,
            modalita: options?.modalita ?? 'unisci',
          });
          vaniAggiunti = res.vaniImportati;
        }

        const mqMatch = `${file.name} ${testo.slice(0, 2000)}`.match(
          /(\d+(?:[.,]\d+)?)\s*(?:m²|mq|m2)\b/i
        );
        const mqRilevati = mqMatch ? parseNumero(mqMatch[1], 0) : 0;

        let vociAggiornate = 0;
        const catLower = (options?.categoria ?? '').toLowerCase();
        const isComputoOrCatasto =
          /computo|preventiv|catast|planimetr|rai|impiant/.test(catLower) ||
          mqRilevati > 0;

        if (mqRilevati > 0 && isComputoOrCatasto) {
          const vociAttuali = get().voci[targetId] ?? [];
          if (vociAttuali.length > 0) {
            set((state) => ({
              voci: {
                ...state.voci,
                [targetId]: (state.voci[targetId] ?? []).map((v) => {
                  if (v.oneriSicurezza) return v;
                  if (v.unitaMisura === 'mq' || v.unitaMisura === 'm²' || v.um === 'mq') {
                    vociAggiornate += 1;
                    const q = Number(mqRilevati.toFixed(2));
                    const p = Number(v.prezzoUnitario) || 0;
                    return {
                      ...v,

export async function caricaDaSupabase() {
  const sb = getSupabase();
  if (!sb) return null;

  const { data: { user } } = await sb.auth.getUser();
  if (!user) return null;

  const { data: commesseRows, error } = await sb
    .from('commesse')
    .select('*')
    .order('created_at', { ascending: false });

  if (error || !commesseRows || commesseRows.length === 0) return null;

  const commesse: Commessa[] = commesseRows.map((r) => ({
    id: r.id,
    codice: r.codice_commessa,
    titolo: r.titolo,

    tipologia: r.tipologia_intervento ?? 'CILA',

    cartaQualita: Boolean(r.carta_qualita),
    vincoloSoprintendenza: Boolean(r.vincoli_note),
    createdAt: r.created_at,
  }));

  const [raiRes, vociRes] = await Promise.all([
    sb.from('verifiche_rai').select('*'),
    sb.from('voci_computo').select('*').order('ordine', { ascending: true }),
  ]);

  const vani: Record<string, Vano[]> = {};
  (raiRes.data ?? []).forEach((v) => {

      superfFinestrata: Number(v.superficie_finestrata_apribile),
      altezzaMedia: Number(v.altezza_media_ponderale),
      vmc: (v.note_tecniche ?? '').includes('VMC'),
    });
  });

  const voci: Record<string, VoceComputo[]> = {};
  (vociRes.data ?? []).forEach((vc) => {

export async function sincronizzaCommessaSuSupabase(
  c: Commessa,
  vaniCommessa: Vano[] = [],
  vociCommessa: VoceComputo[] = []
) {
  const sb = getSupabase();
  if (!sb) throw new Error('Client Supabase non configurato');

  const { data: { user } } = await sb.auth.getUser();
  if (!user) {
    throw new Error('Effettua il login per sincronizzare su Cloud');
  }

  let safeCommessaId = c.id;
  if (!isUuid(c.id)) {
    safeCommessaId = crypto.randomUUID();
    useStore.setState((prev) => ({
      commesse: prev.commesse.map((item) =>
        item.id === c.id ? { ...item, id: safeCommessaId } : item
      ),
      attivaId: prev.attivaId === c.id ? safeCommessaId : prev.attivaId,
      vani: {
        ...prev.vani,
        [safeCommessaId]: prev.vani[c.id] ?? vaniCommessa,
      },
      voci: {
        ...prev.voci,
        [safeCommessaId]: prev.voci[c.id] ?? vociCommessa,
      },
    }));
  }

  const { error: errComm } = await sb.from('commesse').upsert(
    {
      id: safeCommessaId,
      user_id: user.id,
      codice_commessa: c.codice,
      titolo: c.titolo,
      indirizzo: c.indirizzo,
      municipio: c.municipio,
      dati_catastali: {
        foglio: c.foglio,
        particella: c.particella,
        subalterno: c.subalterno,
        categoria: c.categoriaCatastale,
        rendita: c.rendita,
      },
      tipologia_intervento: c.tipologia,
      tessuto_prg: c.tessutoPRG,
      carta_qualita: c.cartaQualita,
      vincoli_note: c.vincoloSoprintendenza ? 'Vincolo D.Lgs 42/2004' : '',
    },
    { onConflict: 'id' }
  );
  if (errComm) throw new Error(`Commessa: ${errComm.message}`);

  const { error: delRaiErr } = await sb
    .from('verifiche_rai')
    .delete()
    .eq('commessa_id', safeCommessaId);
  if (delRaiErr) throw new Error(`Pulizia RAI: ${delRaiErr.message}`);

  if (vaniCommessa.length > 0) {
    const { error: errRai } = await sb.from('verifiche_rai').insert(
      vaniCommessa.map((v) => ({
        id: isUuid(v.id) ? v.id : crypto.randomUUID(),
        commessa_id: safeCommessaId,
        nome_vano: v.nome || 'Vano senza nome',
        destinazione_uso: v.destinazione,
        superficie_calpestabile: v.superficie,
        superficie_finestrata_apribile: v.superfFinestrata,
        altezza_media_ponderale: v.altezzaMedia,
        note_tecniche: v.vmc ? 'VMC attiva' : '',
      }))
    );
    if (errRai) throw new Error(`Vani RAI: ${errRai.message}`);
  }

  const { error: delVociErr } = await sb
    .from('voci_computo')
    .delete()
    .eq('commessa_id', safeCommessaId);
  if (delVociErr) throw new Error(`Pulizia computo: ${delVociErr.message}`);

  if (vociCommessa.length > 0) {
    const { error: errVoci } = await sb.from('voci_computo').insert(
      vociCommessa.map((vc, idx) => ({
        id: isUuid(vc.id) ? vc.id : crypto.randomUUID(),
        commessa_id: safeCommessaId,
        ordine: idx + 1,
        codice_prezzario: vc.codicePrezzario,
        categoria_wbs: vc.wbs,
        descrizione: vc.descrizione,
        unita_misura: vc.um,
        quantita: vc.quantita,
        prezzo_unitario: vc.prezzoUnitario,
        is_sicurezza_non_ribassabile: vc.oneriSicurezza,
      }))
    );
    if (errVoci) throw new Error(`Voci computo: ${errVoci.message}`);
  }
}

function loadDocsForCommessa(commessaId: string): DocumentoArchivio[] {
  try {
    const raw = window.localStorage.getItem(`my-archinext-docs-${commessaId}`);
    if (raw) {

function formatBytes(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
}

function loadCronoForCommessa(commessaId: string): CronoStorage {
  try {
    const raw = window.localStorage.getItem(`my-archinext-crono-${commessaId}`);
    if (raw) {
      const parsed = JSON.parse(raw) as Partial<CronoStorage>;
      if (parsed.dataInizio && Array.isArray(parsed.fasi) && parsed.fasi.length > 0) {
        return { dataInizio: parsed.dataInizio, fasi: parsed.fasi };
      }
    }
  } catch {
    // Fallback su dati iniziali
  }
  return { dataInizio: '2026-12-01', fasi: FASI_INIZIALI };
}

function addWeeks(isoDate: string, weeks: number): Date {
  const base = new Date(`${isoDate}T00:00:00`);
  const safe = Number.isNaN(base.getTime()) ? new Date() : base;
  const result = new Date(safe);
  result.setDate(result.getDate() + Math.round(weeks * 7));
  return result;
}

function formatShortDate(d: Date): string {
  return d.toLocaleDateString('it-IT', { day: '2-digit', month: '2-digit' });
}

export function Cronoprogramma() {
  const { commessa } = useAttiva();
  const commessaId = commessa?.id ?? 'default';

  const [dataInizio, setDataInizio] = useState<string>(
    () => loadCronoForCommessa(commessaId).dataInizio
  );
  const [fasi, setFasi] = useState<FaseCantiere[]>(() => loadCronoForCommessa(commessaId).fasi);
  const [nuovaAttivita, setNuovaAttivita] = useState('');
  const [nuoveSettimane, setNuoveSettimane] = useState(2);

  useEffect(() => {
    const loaded = loadCronoForCommessa(commessaId);
    setDataInizio(loaded.dataInizio);
    setFasi(loaded.fasi);
  }, [commessaId]);

    try {
      window.localStorage.setItem(
        `my-archinext-crono-${commessaId}`,
        JSON.stringify({ dataInizio, fasi })
      );
    } catch {
      // Ignora eventuali limiti di storage
    }
  }, [commessaId, dataInizio, fasi]);

  const calcolate = useMemo(() => {
    let cursorWeeks = 0;
    return fasi.map((fase, idx) => {
      const dur = Math.max(0.5, Number(fase.settimane) || 1);
      const startWeek = cursorWeeks;
      const endWeek = cursorWeeks + dur;
      const startDate = addWeeks(dataInizio, startWeek);
      const endDate = addWeeks(dataInizio, endWeek);
      cursorWeeks = endWeek;
      return {
        ...fase,
        wbs: String(idx + 1).padStart(2, '0'),
        dur,
        startWeek,
        endWeek,
        startDate,
        endDate,
      };
    });
  }, [fasi, dataInizio]);

  const settimaneTotali = useMemo(
    () => calcolate.reduce((acc, f) => acc + f.dur, 0),
    [calcolate]
  );

interface GeoPoint {

  lon: number;

function getFallbackCoords(indirizzo: string): GeoPoint {
  const clean = indirizzo.trim();
  const lower = clean.toLowerCase();
  if (lower.includes('boschetto')) {
    return { lat: 41.8955, lon: 12.4912, label: 'Rione Monti · Via del Boschetto' };
  }
  if (lower.includes('termini')) {
    return { lat: 41.9009, lon: 12.502, label: 'Roma Termini · Esquilino' };
  }
  return { lat: 41.8902, lon: 12.4922, label: clean || 'Roma Capitale' };
}

function parseStreetAndNumber(indirizzo: string): { streetQuery: string; cap?: string } {
  const firstPart = indirizzo.split(',')[0]?.trim() || indirizzo.trim();
  const capMatch = indirizzo.match(/\b(00\d{3})\b/);
  const cap = capMatch ? capMatch[1] : undefined;

  const parts = indirizzo.split(',').map((p) => p.trim());
  if (parts.length >= 2 && /^\d+[a-zA-Z]?$/.test(parts[1])) {
    return { streetQuery: `${parts[1]} ${parts[0]}`, cap };
  }
  const numMatch = firstPart.match(/^(.*?)\s+(\d+[a-zA-Z]?)$/);
  if (numMatch) {
    return { streetQuery: `${numMatch[2]} ${numMatch[1]}`, cap };
  }
  return { streetQuery: firstPart, cap };
}

    if (pianoInduzione) potenzaKw += 1.5;
    if (pompaCalore) potenzaKw += 1.5;
    if (livelloCEI === '2') potenzaKw = Math.max(potenzaKw, 4.5);
    if (livelloCEI === '3') potenzaKw = Math.max(potenzaKw, 6.0);

    const puntiPreseMin = Math.max(18, Math.round(mqTotali * 0.35) + nCamere * 3);
    const puntiLuceMin = Math.max(8, vani.length > 0 ? vani.length * 2 : 10);
    const circuitiMin =
      livelloCEI === '1'
        ? mqTotali <= 50
          ? 2
          : mqTotali <= 75
            ? 3
            : 4
        : livelloCEI === '2'
          ? 5
          : 6;

    const potenzaTermicaKw = Number(((mqTotali * 3.0 * 35) / 1000).toFixed(1));
    const obbligoProgettoProfessionista = potenzaKw > 6.0 || mqTotali > 400;

    return {
      mqTotali,
      nBagni,
      potenzaKw: Number(potenzaKw.toFixed(1)),
      puntiPreseMin,
      puntiLuceMin,
      circuitiMin,
      potenzaTermicaKw,
      obbligoProgettoProfessionista,
    };
  }, [vani, livelloCEI, pianoInduzione, pompaCalore]);

  if (!commessa) {
    return <div className="p-10 text-zinc-600">Nessuna commessa attiva.</div>;
  }

  return (
    <div className="space-y-4">
      <header>
        <h1 className="text-xl sm:text-2xl font-bold tracking-tight text-zinc-950">
          5. Dimensionamento Impianti (CEI 64-8 / UNI 7129)
        </h1>
        <p className="text-xs sm:text-sm text-zinc-700 mt-0.5">
          Dotazioni minime impianto elettrico, fabbisogno termico e adempimenti DM 37/2008. Commessa:{' '}
          <strong className="text-zinc-950">{commessa.codice}</strong>
        </p>
      </header>

      {/* Parametri configurazione */}
      <section className="rounded-xl bg-white border border-zinc-300 p-5 space-y-4">
        <div className="flex flex-wrap items-center justify-between gap-3">
          <div>
            <h2 className="text-sm font-bold text-zinc-950">Configurazione Impiantistica</h2>
            <p className="text-xs text-zinc-600">
              Calibrata su {calcolo.mqTotali.toFixed(1)} mq ({vani.length || 5} vani censiti)
            </p>
          </div>

          <div className="inline-flex rounded-lg bg-zinc-100 p-1 border border-zinc-300 text-xs font-mono">
            {(['1', '2', '3'] as const).map((lvl) => (
              <button
                key={lvl}
                type="button"
                onClick={() => setLivelloCEI(lvl)}
                className={`px-3 py-1.5 rounded-md transition-colors ${
                  livelloCEI === lvl
                    ? 'bg-zinc-950 text-white font-semibold'
                    : 'text-zinc-700 hover:text-zinc-950'
                }`}
              >
                Livello {lvl} CEI
              </button>
            ))}
          </div>
        </div>

        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 pt-1">
          <label className="flex items-center justify-between rounded-lg border border-zinc-300 bg-zinc-50 px-3.5 py-2.5 cursor-pointer">
            <span className="text-xs font-semibold text-zinc-900">Piano cottura a induzione</span>
            <input
              type="checkbox"
              checked={pianoInduzione}
              onChange={(e) => setPianoInduzione(e.target.checked)}
              className="w-4 h-4 accent-zinc-950"
            />
          </label>
          <label className="flex items-center justify-between rounded-lg border border-zinc-300 bg-zinc-50 px-3.5 py-2.5 cursor-pointer">
            <span className="text-xs font-semibold text-zinc-900">Climatizzazione / Pompa di calore</span>
            <input
              type="checkbox"
              checked={pompaCalore}
              onChange={(e) => setPompaCalore(e.target.checked)}
              className="w-4 h-4 accent-zinc-950"
            />
          </label>
        </div>
      </section>

      {/* Schede risultati */}
      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
        <section className="rounded-xl bg-zinc-950 text-zinc-100 border border-zinc-800 p-5 space-y-3">
          <div className="flex items-center gap-2 pb-2.5 border-b border-zinc-800">
            <Zap size={16} className="text-amber-300" />
            <h3 className="text-xs font-mono font-semibold uppercase tracking-wider text-white">
              Impianto Elettrico — CEI 64-8 Cap. 37
            </h3>
          </div>

          <div className="grid grid-cols-2 gap-2.5 text-xs">
            <div className="rounded-lg bg-zinc-900 border border-zinc-800 p-3">
              <span className="block text-[10px] font-mono uppercase text-zinc-400">
                Potenza Impegnata
              </span>
              <span className="block text-lg font-mono font-bold text-white mt-0.5">
                {calcolo.potenzaKw} kW
              </span>
            </div>
            <div className="rounded-lg bg-zinc-900 border border-zinc-800 p-3">
              <span className="block text-[10px] font-mono uppercase text-zinc-400">
                Circuiti Minimi
              </span>
              <span className="block text-lg font-mono font-bold text-white mt-0.5">
                {calcolo.circuitiMin} linee
              </span>
            </div>
            <div className="rounded-lg bg-zinc-900 border border-zinc-800 p-3">
              <span className="block text-[10px] font-mono uppercase text-zinc-400">
                Punti Prese Min.
              </span>
              <span className="block text-base font-mono font-bold text-white mt-0.5">
                ≥ {calcolo.puntiPreseMin}
              </span>
            </div>
            <div className="rounded-lg bg-zinc-900 border border-zinc-800 p-3">
              <span className="block text-[10px] font-mono uppercase text-zinc-400">
                Punti Luce Min.
              </span>
              <span className="block text-base font-mono font-bold text-white mt-0.5">
                ≥ {calcolo.puntiLuceMin}
              </span>
            </div>
          </div>

          <p className="text-[11px] text-zinc-300 leading-relaxed">
            Prevedere almeno 2 interruttori differenziali (30 mA tipo A/F) per garantire continuità di servizio e SPD contro sovratensioni.
          </p>
        </section>

        <section className="rounded-xl bg-zinc-950 text-zinc-100 border border-zinc-800 p-5 space-y-3">
          <div className="flex items-center gap-2 pb-2.5 border-b border-zinc-800">
            <Flame size={16} className="text-emerald-300" />
            <h3 className="text-xs font-mono font-semibold uppercase tracking-wider text-white">
              Termotecnica &amp; Adempimenti DM 37/08
            </h3>
          </div>

          <div className="grid grid-cols-2 gap-2.5 text-xs">
            <div className="rounded-lg bg-zinc-900 border border-zinc-800 p-3">
              <span className="block text-[10px] font-mono uppercase text-zinc-400">
                Fabbisogno Termico
              </span>
              <span className="block text-lg font-mono font-bold text-white mt-0.5">
                ~ {calcolo.potenzaTermicaKw} kW
              </span>
            </div>
            <div className="rounded-lg bg-zinc-900 border border-zinc-800 p-3">
              <span className="block text-[10px] font-mono uppercase text-zinc-400">
                Adduzione Gas
              </span>
              <span className="block text-sm font-mono font-bold text-white mt-1">
                {pianoInduzione && pompaCalore ? 'Gas-Free' : 'UNI 7129'}
              </span>
            </div>
          </div>

          <div className="rounded-lg bg-zinc-900 border border-zinc-800 p-3 text-xs">
            {calcolo.obbligoProgettoProfessionista ? (
              <div className="flex items-start gap-2 text-amber-300">
                <AlertTriangle size={15} className="shrink-0 mt-0.5" />
                <span>
                  Potenza &gt; 6 kW: obbligatorio progetto firmato da professionista iscritto all&apos;Albo (DM 37/2008 art. 5).
                </span>
              </div>
            ) : (
              <div className="flex items-start gap-2 text-emerald-300">
                <CheckCircle2 size={15} className="shrink-0 mt-0.5" />
                <span>
                  Potenza ≤ 6 kW: progetto a cura del Responsabile Tecnico dell&apos;impresa installatrice e rilascio Di.Co.
                </span>
              </div>
            )}
          </div>

          <div className="flex items-center gap-1.5 text-[11px] font-mono text-zinc-400">
            <ShieldCheck size={13} />
            <span>Verificare scarico fumi a tetto (DPR 412/93) in Città Storica T1.</span>
          </div>
        </section>
      </div>
    </div>
  );
}
